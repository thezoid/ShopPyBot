PLUGIN_API_VERSION = 2  # bumped from 1

class RetailerPlugin(ABC):
    domain_patterns: list[str]
    def __init__(self, config) -> None: self.config = config; self.driver = None
    async def setup(self) -> None: ...
    @abstractmethod
    async def check_availability(self, url: str) -> bool: ...
    @abstractmethod
    async def auto_buy(self, url: str) -> bool: ...
    async def login(self) -> None: ...        # no-op default, returns None
    async def detect_captcha(self) -> bool: ...  # no-op default, returns False
    async def teardown(self) -> None: ...

Current tests/conftest.py fixtures (preserve): tmp_config_yml, tmp_data_dir.
pyproject.toml already sets asyncio_mode = "auto" and asyncio_default_fixture_loop_scope = "function".
